!
end
!

!

vlan 30

# SW-ACCESS 1

en
conf t
!
no ip domain-lookup
!
hostname SW-ACCESS1
!
line con 0
 logging syn
 exec-timeout 0 0
 password REPLACE_SECRET_5
 login
!
conf t
!
spanning-tree mode rapid-pvst
!
vtp password REPLACE_SECRET_4
vtp domain HQ
vtp version 3
vtp mode client
!
interface range gi0/0 - 1
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40,99
 no sh
!
interface gi0/0
 spanning-tree vlan 10 cost 10 # Cost Down 으로 인해 우선 경로 지정
interface gi0/1
 spanning-tree vlan 10 cost 100 # Cost Up 으로 인해 백업 경로 지정 
!
interface range gi0/2-3
 switchport mode access
 switchport access vlan 10
 spanning-tree bpduguard enable # Loop 방지를 위해서 BPDU 수신시 포트 차단
 spanning-tree portfast # 최적화를 위해서 스패닝트리 대기 과정 없이 Forwarding 상태로 전환
 no sh
!
end
!
